{{- /* An RSS 2.0 feed of listed posts, as src/blog/rss.ts wrote it: .posts (newest first; at most 50 are used) and, for a tag's feed, .tag. Addresses are absolute against the site's base URL, like the pages' canonical links. Returns the XML as a string (the blogrss output format is plain text, so it is written as it is). */ -}} {{- $base := site.BaseURL | strings.TrimSuffix "/" -}} {{- $name := "Vault Learning Games" -}} {{- $x := "blog/xml.html" -}} {{- $title := printf "%s Blog" $name }}{{ $page := printf "%s/blog" $base }}{{ $self := printf "%s/blog/rss.xml" $base -}} {{- $desc := printf "Posts from the %s blog." $name -}} {{- with .tag -}} {{- $title = printf "%s — %s" $title .name }}{{ $page = printf "%s/blog/tag/%s" $base .slug }}{{ $self = printf "%s/blog/tag/%s/rss.xml" $base .slug -}} {{- $desc = printf "Posts tagged \"%s\" on the %s blog." .name $name -}} {{- end -}} {{- $posts := first 50 (.posts | default slice) -}} {{- $rfc := "Mon, 02 Jan 2006 15:04:05 GMT" -}} {{- $lines := slice "" "" "" (printf "%s" (partial $x $title)) (printf "%s" (partial $x $page)) (printf "%s" (partial $x $desc)) -}} {{- /* The newest post's date; a feed with no posts has no lastBuildDate, so it is the same at every build. */ -}} {{- with $posts }}{{ with (index . 0).published_at }}{{ $lines = $lines | append (printf "%s" ((time .).UTC.Format $rfc)) }}{{ end }}{{ end -}} {{- $lines = $lines | append (printf "" (partial $x $self)) -}} {{- range $posts -}} {{- $url := printf "%s/blog/%s" $base .slug -}} {{- $lines = $lines | append "" (printf "%s" (partial $x .title)) (printf "%s" (partial $x $url)) (printf "%s" (partial $x $url)) -}} {{- with .published_at }}{{ $lines = $lines | append (printf "%s" ((time .).UTC.Format $rfc)) }}{{ end -}} {{- with .author.name }}{{ $lines = $lines | append (printf "%s" (partial $x .)) }}{{ end -}} {{- $lines = $lines | append (printf "%s" (partial $x .description)) -}} {{- range .tags }}{{ $lines = $lines | append (printf "%s" (partial $x .name)) }}{{ end -}} {{- with .featured_image }}{{ if .type -}} {{- $img := .url }}{{ if hasPrefix $img "/" }}{{ $img = printf "%s%s" $base $img }}{{ end -}} {{- $lines = $lines | append (printf "" (partial $x $img) .type) -}} {{- end }}{{ end -}} {{- $lines = $lines | append "" -}} {{- end -}} {{- $lines = $lines | append "" "" "" -}} {{- return delimit $lines "\n" -}}
