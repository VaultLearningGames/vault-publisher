{{- /* The topics to filter the blog by, on /blog and /blog/tag/ (.active: the tag's slug there, "" on /blog): "All", then the tags with a listed post, most posts first (ties by name), as pills. The first 8 are in the row and the rest behind "+N more" (a
, no script); the active tag is always in the row, in the 8th place if it would have been behind "+N more". Nothing when no tag has a listed post. */ -}} {{- $active := .active | default "" -}} {{- $blog := hugo.Data.blog | default dict -}} {{- $listed := where ($blog.posts | default slice) "listed" true -}} {{- $keyed := slice -}} {{- range $blog.tags | default slice -}} {{- $n := len (where $listed "slug" "in" (.post_slugs | default slice)) -}} {{- if gt $n 0 }}{{ $keyed = $keyed | append (dict "tag" . "key" (printf "%09d|%s" (sub 999999999 $n) (lower .name))) }}{{ end -}} {{- end -}} {{- $tags := slice }}{{ range sort $keyed "key" }}{{ $tags = $tags | append .tag }}{{ end -}} {{- $max := 8 -}} {{- $row := $tags }}{{ $more := slice -}} {{- if gt (len $tags) $max -}} {{- $row = first $max $tags }}{{ $more = after $max $tags -}} {{- with where $more "slug" $active -}} {{- $row = (first (sub $max 1) $tags) | append (index . 0) -}} {{- $more = where (after (sub $max 1) $tags) "slug" "ne" $active -}} {{- end -}} {{- end -}} {{- with $tags }}
  • All

  • {{ partial "blog/pill.html" (dict "tag" . "active" $active) }}
    • {{- end }}
{{- with $more }}
+{{ len . }} more
    {{- range . }}
  • {{ partial "blog/pill.html" (dict "tag" . "active" $active) }}
