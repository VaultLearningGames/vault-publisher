var VAULT_CHANNELS = { 'https://pbskids.org': 'pbskids.org', 'https://www.pbskids.org': 'pbskids.org', 'https://pbswisconsineducation.org': 'pbswisconsineducation.org', 'https://www.pbswisconsineducation.org': 'pbswisconsineducation.org', 'https://pbslearningmedia.org': 'pbslearningmedia.org', 'https://www.pbslearningmedia.org': 'pbslearningmedia.org', 'https://sciencegamecenter.org': 'sciencegamecenter.org', 'https://www.sciencegamecenter.org': 'sciencegamecenter.org' }; // Short keys partner links already carry, mapped to the hostname we report them as. var VAULT_CHANNEL_ALIASES = { pbswi: 'pbswisconsineducation.org', pbskids: 'pbskids.org', pbslm: 'pbslearningmedia.org', sciencegamecenter: 'sciencegamecenter.org' }; // The partner hostnames are real referrals, so they are reported with the referral medium analytics groups // them under. "vault", "other" and direct plays are not referrals and keep the honest "vault-player". var VAULT_PARTNER_CHANNELS = Object.keys(VAULT_CHANNEL_ALIASES).map(function (k) { return VAULT_CHANNEL_ALIASES[k]; }); function vaultResolveChannel(rawCh, source, embedded) { var channel = rawCh || ''; // An explicit vault_ch is reported exactly as given, so the short keys are normalised here — right where // the value is read, before anything is reported, and not only on the VAULT_CHANNELS fallback below. if (Object.prototype.hasOwnProperty.call(VAULT_CHANNEL_ALIASES, channel)) channel = VAULT_CHANNEL_ALIASES[channel]; // No channel, but a source we recognise: name it anyway. if (!channel && /^https:\/\//.test(source)) channel = VAULT_CHANNELS[new URL(source).origin] || 'other'; // Inside the Vault player with no outside source: the play came through Vault itself. if (!channel && embedded) channel = 'vault'; return { channel: channel, medium: VAULT_PARTNER_CHANNELS.indexOf(channel) !== -1 ? 'referral' : 'vault-player' }; } // Analytics.1b: report the resolved channel to whichever analytics the page actually has. gtag and ga are // checked when this runs, not when the script loads, so an analytics script that loads after this file still // receives the report. The fields are the reference's own, in its own order: utm_source / utm_medium carried // on the reported location (analytics prefers them over the referrer and never drops them as a self-referral) // and dimension1, the "Play source" custom dimension. The reference's ga('set', field, value) calls become // gtag's gtag('set', { field: value }) form, nothing else. A falsy channel (a direct play) reports nothing, // and a page with no analytics at all gets no call, no console output, no throw. function vaultReportChannel(channel) { if (!channel) return; try { if (typeof window.gtag === 'function') { var medium = VAULT_PARTNER_CHANNELS.indexOf(channel) !== -1 ? 'referral' : 'vault-player'; var reported = new URL(location.href); reported.searchParams.set('utm_source', channel); reported.searchParams.set('utm_medium', medium); window.gtag('set', { location: reported.href }); window.gtag('set', { dimension1: channel }); // registered as "Play source" on the property } else if (typeof window.ga === 'function') { window.ga('set', 'dimension1', channel); // registered as "Play source" on the property } } catch (e) {} }
